get_filename_component(SCL_UTILITY_TEST_DIR "${CMAKE_CURRENT_LIST_DIR}/../../../test" ABSOLUTE)

if (EXISTS "${SCL_UTILITY_TEST_DIR}")
    file(GLOB SCL_UTILITY_TEST_SUBDIRS LIST_DIRECTORIES true "${SCL_UTILITY_TEST_DIR}/*")

    set(compile_fail_requires_values msvc_abi msvc clang gcc gnu)
    set(compile_fail_requires_met "")
    if (MSVC OR CMAKE_CXX_SIMULATE_ID STREQUAL "MSVC")
        list(APPEND compile_fail_requires_met msvc_abi)
    endif()
    if (CMAKE_CXX_COMPILER_ID STREQUAL "MSVC")
        list(APPEND compile_fail_requires_met msvc)
    elseif (CMAKE_CXX_COMPILER_ID MATCHES "Clang")
        list(APPEND compile_fail_requires_met clang gnu)
    elseif (CMAKE_CXX_COMPILER_ID STREQUAL "GNU")
        list(APPEND compile_fail_requires_met gcc gnu)
    endif()

    foreach(subdir IN LISTS SCL_UTILITY_TEST_SUBDIRS)
        if (IS_DIRECTORY "${subdir}")
            get_filename_component(name "${subdir}" NAME)

            # --- Companion shared library ---
            # Without a test framework no test target exists to link the library into.
            set(common_libs "")
            if (GTEST_FOUND OR DOCTEST_FOUND OR CATCH2_FOUND)
                file(GLOB_RECURSE shared_srcs CONFIGURE_DEPENDS "${subdir}/*_shared.cpp")
                if (shared_srcs)
                    set(shared_lib "utility_${name}_shared")
                    add_library(${shared_lib} SHARED ${shared_srcs})
                    target_link_libraries(${shared_lib} PRIVATE scl::utility)
                    target_include_directories(${shared_lib} PRIVATE "${SCL_UTILITY_TEST_DIR}")
                    list(APPEND common_libs ${shared_lib})
                endif()
            endif()

            # --- Builds that must fail ---
            file(GLOB_RECURSE fail_srcs CONFIGURE_DEPENDS "${subdir}/*_compile_fail.cpp")
            foreach(src IN LISTS fail_srcs)
                file(STRINGS "${src}" expect REGEX "^// expect: " LIMIT_COUNT 1)
                file(STRINGS "${src}" requires REGEX "^// requires: " LIMIT_COUNT 1)
                file(STRINGS "${src}" options REGEX "^// options: " LIMIT_COUNT 1)
                string(REPLACE "// expect: " "" expect "${expect}")
                string(REPLACE "// requires: " "" requires "${requires}")
                string(REPLACE "// options: " "" options "${options}")
                if (NOT expect)
                    message(FATAL_ERROR "${src} has no line `// expect: <regular expression>`")
                endif()
                if (requires AND NOT requires IN_LIST compile_fail_requires_values)
                    message(FATAL_ERROR "${src} requires an unknown compiler: ${requires}")
                elseif (requires AND NOT requires IN_LIST compile_fail_requires_met)
                    continue()
                endif()
                get_filename_component(stem "${src}" NAME_WE)
                set(tgt "utility_${name}_${stem}")
                add_library(${tgt} OBJECT EXCLUDE_FROM_ALL "${src}")
                target_link_libraries(${tgt} PRIVATE scl::utility)
                target_compile_features(${tgt} PRIVATE cxx_std_20)
                if (options)
                    target_compile_options(${tgt} PRIVATE "SHELL:${options}")
                endif()
                add_test(NAME ${tgt}
                    COMMAND "${CMAKE_COMMAND}" "-DBUILD_DIR=${CMAKE_BINARY_DIR}" "-DTARGET=${tgt}"
                        "-DCONFIG=$<CONFIG>" "-DPATTERN=${expect}" -P "${CMAKE_CURRENT_LIST_DIR}/compile_fail.cmake"
                )
                set_tests_properties(${tgt} PROPERTIES RUN_SERIAL TRUE)
            endforeach()

            # --- GoogleTest ---
            if (GTEST_FOUND)
                file(GLOB_RECURSE srcs CONFIGURE_DEPENDS
                    "${subdir}/*_gtest.c" "${subdir}/*_gtest.cc" "${subdir}/*_gtest.cpp"
                )
                if (srcs)
                    set(tgt "utility_${name}_gtest")
                    add_executable(${tgt} ${srcs})
                    target_link_libraries(${tgt} PRIVATE scl::utility GTest::gtest_main ${common_libs})
                    target_compile_features(${tgt} PRIVATE cxx_std_20)
                    target_include_directories(${tgt} PRIVATE "${SCL_UTILITY_TEST_DIR}")
                    include(GoogleTest)
                    gtest_discover_tests(${tgt} DISCOVERY_TIMEOUT 30 DISCOVERY_MODE PRE_TEST)
                endif()
            endif()

            # --- doctest ---
            if (DOCTEST_FOUND)
                file(GLOB_RECURSE srcs CONFIGURE_DEPENDS
                    "${subdir}/*_doctest.c" "${subdir}/*_doctest.cc" "${subdir}/*_doctest.cpp"
                )
                if (srcs)
                    set(tgt "utility_${name}_doctest")
                    add_executable(${tgt} ${srcs})
                    target_link_libraries(${tgt} PRIVATE scl::utility doctest::doctest ${common_libs})
                    target_compile_features(${tgt} PRIVATE cxx_std_20)
                    add_test(NAME ${tgt} COMMAND ${tgt})
                endif()
            endif()

            # --- Catch2 ---
            if (CATCH2_FOUND)
                file(GLOB_RECURSE srcs CONFIGURE_DEPENDS
                    "${subdir}/*_catch2.c" "${subdir}/*_catch2.cc" "${subdir}/*_catch2.cpp"
                )
                if (srcs)
                    if (TARGET Catch2::Catch2WithMain)
                        set(tgt "utility_${name}_catch2")
                        add_executable(${tgt} ${srcs})
                        target_link_libraries(${tgt} PRIVATE scl::utility Catch2::Catch2WithMain ${common_libs})
                        target_compile_features(${tgt} PRIVATE cxx_std_20)
                        include(Catch)
                        catch_discover_tests(${tgt})
                    elseif (TARGET Catch2::Catch2)
                        set(tgt "utility_${name}_catch2_v2")
                        add_executable(${tgt} ${srcs})
                        target_link_libraries(${tgt} PRIVATE scl::utility Catch2::Catch2 ${common_libs})
                        target_compile_features(${tgt} PRIVATE cxx_std_20)
                        add_test(NAME ${tgt} COMMAND ${tgt})
                    endif()
                endif()
            endif()
        endif()
    endforeach()
endif()
